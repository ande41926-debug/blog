```python
# Dependency Imports
import pandas as pd
import matplotlib.pyplot as plt
import json
from js import fetch
```

```python
# Helper function which uses the browser's built-in fetch API
async def get_json(url):
    resp = await fetch(url)
    txt = await resp.text()
    return json.loads(txt)
```

```python
# Data Acquisition by using the helper function
data = await get_json(
    "https://api.census.gov/data/2026/cps/basic/feb?tabulate=weight(PWSSWGT)&col+PEMLR&row+PEMARITL&for=state:01&PEEDUCA=39&key=YOUR_API_KEY"
)
```

```python
# Data Wrangling a.k.a. Data Munging
# Parse the tabulation into a tidy DataFrame
header = data[0]
rows = data[1:]

col_codes = [item["PEMLR"] for item in header[:-1]]
row_order = [row[-1] for row in rows]

df = pd.DataFrame([r[:-1] for r in rows], columns=col_codes, index=row_order)
df = df.apply(pd.to_numeric, errors="coerce").fillna(0)
```

```python
# Label maps
pemlr_labels = {
    "-1": "Not in universe",
    "1": "Employed",
    "2": "Unemployed",
    "3": "Unemployed, on layoff",
    "4": "Not in labor force",
    "5": "Unknown",
    "6": "Armed forces",
    "7": "Other"
}

pemaritl_labels = {
    "1": "Married, spouse present",
    "2": "Married, spouse absent",
    "3": "Widowed",
    "4": "Divorced",
    "5": "Separated",
    "6": "Never married"
}

df.index = [pemaritl_labels.get(str(x), str(x)) for x in df.index]
df.columns = [pemlr_labels.get(str(x), str(x)) for x in df.columns]
```

```python
# Sort by total (descending)
df = df.loc[df.sum(axis=1).sort_values(ascending=False).index]
```

```python
# Prepare for Data Visualization
# Graph 1: Average weighted count by labor force status
summary = df.groupby("PEMLR", as_index=False)["PWSSWGT"].mean().sort_values("PWSSWGT", acsending=False)
summary["Label"] = summary["PEMLR"].map(pemlr_labels.get(int(x), str(x)))

fig, ax = plt.subplots(figsize=(12, 7))
ax.bar(summary["label], summary["PWSSWGT"], color="steelblue")
ax.set_title("Average PWSSWGT by Labor force Status")
ax.set_xlabel("Labor Force status(PEMLR)")
ax.set_ylabel("Average PWSSWGT")
ax.legend(title="PEMLR", bbox_to_anchor=(1.02, 1), loc="upper left")
plt.xticks(rotation=30, ha="right")
plt.tight_layout()
plt.show()

# Graph 2: Top 10 states by average wheighted count
state_summary = df.groupby("state", as_index=False)["PWSSWGT"].mean().sort_values("PWSSWGT", ascending=False).head(10)

fig, ax = plt.subplots(figsize=(10,6))
ax.bar(state_summary["state"].astype(str), state_summary["PWSSWGT"], ascending=False).head(10)
```
